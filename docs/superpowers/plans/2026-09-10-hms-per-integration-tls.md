# Per-Integration TLS for HMS Adapters — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give every HMS integration its own TLS posture — its own trusted CA, its own client certificate (mTLS), and its own timeout — instead of the process-wide `SSL_CERT_FILE` workaround and the hard-coded 30s timeout.

**Architecture:** A new `sm_common/integrations/tls.py` turns TLS material held in an integration's (already-encrypted) `credentials_json` into an `ssl.SSLContext`. `build_adapter` parses that material out of the credentials dict and hands a `TlsConfig` to each of the four adapters, which pass it to their single `httpx.AsyncClient`. Because `auth.py` signs tokens through the *passed* client and `health_check` uses `self._client`, both the token endpoint and the portal's Test Connection button inherit the same TLS posture automatically.

**Tech Stack:** Python 3.13+, httpx, stdlib `ssl`, `cryptography>=46` (already a dependency), pytest/pytest-asyncio, Caddy 2.11-alpine, Docker Compose.

**Spec:** `docs/superpowers/specs/2026-09-10-hms-tls-and-onboarding-design.md`

**Repo:** `spatiamed-common` (work in the `sm-common-openemr` worktree)

## Global Constraints

- **TLS verification is NEVER disabled.** There is no `insecure`/`verify=False` option anywhere in this plan. A hospital with a private CA supplies `ca_bundle_pem`. Reject any implementation that adds a bypass.
- **All Python commands use `uv run --no-sync ...`** — plain `uv run` silently reverts the editable overlay.
- Ruff config is `select = ["E","F"]` with E402 enforced: **no imports below module-level code**.
- mypy runs strict. Annotate every new function fully. `dict` needs `# type: ignore[type-arg]` where the codebase already does this.
- Line length 100 (ruff default in this repo).
- Timeout is clamped to **≤ 60s**: the poll worker runs a 30s cycle, so an unclamped tenant value would stack requests.
- Never commit private keys, certificates or `.pem` files. CI runs full-history gitleaks. Test certs are generated **at runtime in the test**, never checked in. The Caddy harness certs are generated by a script into a **gitignored** directory.

```python
def _make_cert(common_name: str, *, ca: bool = False) -> tuple[str, str]:
    """Return (cert_pem, key_pem) for a self-signed certificate."""
    key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
    subject = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, common_name)])
    now = datetime.datetime.now(datetime.timezone.utc)
    builder = (
        x509.CertificateBuilder()
        .subject_name(subject)
        .issuer_name(subject)
        .public_key(key.public_key())
        .serial_number(x509.random_serial_number())
        .not_valid_before(now - datetime.timedelta(minutes=5))
        .not_valid_after(now + datetime.timedelta(days=1))
        .add_extension(x509.BasicConstraints(ca=ca, path_length=None), critical=True)
    )
    cert = builder.sign(key, hashes.SHA256())
    cert_pem = cert.public_bytes(serialization.Encoding.PEM).decode()
    key_pem = key.private_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PrivateFormat.PKCS8,
        encryption_algorithm=serialization.NoEncryption(),
    ).decode()
    return cert_pem, key_pem
```

```python
"""Per-integration TLS configuration for HMS adapters.

Each hospital gets its own trust anchor, its own client certificate and its
own timeout. The alternative — the process-wide SSL_CERT_FILE environment
variable — puts every tenant's CA in one shared bundle, so one hospital's
private CA would become trusted for every other hospital's connections.

Verification is never disabled. A hospital running a private CA supplies
``ca_bundle_pem``; that is the supported path and there is no bypass.
"""

from __future__ import annotations

import os
import ssl
import tempfile
from dataclasses import dataclass

DEFAULT_TIMEOUT_SECONDS = 30.0
MAX_TIMEOUT_SECONDS = 60.0
MIN_TIMEOUT_SECONDS = 1.0

TLS_CREDENTIAL_KEYS = (
    "ca_bundle_pem",
    "client_cert_pem",
    "client_key_pem",
    "timeout_seconds",
)


def _clean(value: object) -> str | None:
    """Return a non-blank string, or None. The portal posts "" for untouched fields."""
    if not isinstance(value, str):
        return None
    stripped = value.strip()
    return stripped or None


@dataclass(frozen=True)
class TlsConfig:
    ca_bundle_pem: str | None = None
    client_cert_pem: str | None = None
    client_key_pem: str | None = None
    timeout_seconds: float = DEFAULT_TIMEOUT_SECONDS

    @property
    def is_default(self) -> bool:
        """True when no TLS material is configured and the timeout is untouched."""
        return (
            self.ca_bundle_pem is None
            and self.client_cert_pem is None
            and self.client_key_pem is None
            and self.timeout_seconds == DEFAULT_TIMEOUT_SECONDS
        )

    @classmethod
    def from_credentials(cls, creds: dict) -> TlsConfig:  # type: ignore[type-arg]
        raw_timeout = (creds or {}).get("timeout_seconds")
        if raw_timeout is None:
            timeout = DEFAULT_TIMEOUT_SECONDS
        else:
            try:
                timeout = float(raw_timeout)
            except (TypeError, ValueError):
                # A misconfigured timeout must not take the integration down;
                # fall back to the default rather than raising at build time.
                timeout = DEFAULT_TIMEOUT_SECONDS
            else:
                timeout = min(max(timeout, MIN_TIMEOUT_SECONDS), MAX_TIMEOUT_SECONDS)
        return cls(
            ca_bundle_pem=_clean((creds or {}).get("ca_bundle_pem")),
            client_cert_pem=_clean((creds or {}).get("client_cert_pem")),
            client_key_pem=_clean((creds or {}).get("client_key_pem")),
            timeout_seconds=timeout,
        )


def build_tls_context(cfg: TlsConfig) -> ssl.SSLContext | bool:
    """Turn a TlsConfig into something httpx accepts as ``verify=``.

    Returns True (httpx's own default verification against the system trust
    store) when no TLS material is configured, so integrations that need
    nothing pay for nothing.
    """
    if cfg.ca_bundle_pem is None and cfg.client_cert_pem is None and cfg.client_key_pem is None:
        return True

    if cfg.client_cert_pem is not None and cfg.client_key_pem is None:
        raise ValueError("client_cert_pem was configured without client_key_pem")
    if cfg.client_key_pem is not None and cfg.client_cert_pem is None:
        raise ValueError("client_key_pem was configured without client_cert_pem")

    # create_default_context() loads the system roots, so a hospital's private
    # CA is ADDED to public trust rather than replacing it: the same process
    # still reaches ordinary hosted FHIR servers.
    ctx = ssl.create_default_context()

    if cfg.ca_bundle_pem is not None:
        try:
            ctx.load_verify_locations(cadata=cfg.ca_bundle_pem)
        except ssl.SSLError as exc:
            raise ValueError(f"ca_bundle_pem is not a valid PEM certificate: {exc}") from exc

    if cfg.client_cert_pem is not None and cfg.client_key_pem is not None:
        # CPython's ssl module has no in-memory client-key path:
        # load_cert_chain takes file paths only. The key therefore touches
        # disk for the duration of this call and is removed immediately; the
        # loaded context keeps it in memory thereafter.
        fd, path = tempfile.mkstemp(suffix=".pem")
        try:
            os.fchmod(fd, 0o600)
            with os.fdopen(fd, "w") as handle:
                handle.write(cfg.client_cert_pem)
                if not cfg.client_cert_pem.endswith("\n"):
                    handle.write("\n")
                handle.write(cfg.client_key_pem)
            try:
                ctx.load_cert_chain(certfile=path)
            except ssl.SSLError as exc:
                raise ValueError(f"client certificate or key is not valid PEM: {exc}") from exc
        finally:
            os.unlink(path)

    return ctx
```

```python
"""Runtime certificate generation for TLS tests.

Certificates are generated in-process and never written to the repository:
CI runs full-history gitleaks, and a committed key is a finding forever.
"""

from __future__ import annotations

import datetime
import ipaddress
from dataclasses import dataclass

from cryptography import x509
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.primitives.asymmetric.rsa import RSAPrivateKey
from cryptography.x509.oid import NameOID


@dataclass(frozen=True)
class CertPair:
    cert_pem: str
    key_pem: str


def _serialize(cert: x509.Certificate, key: RSAPrivateKey) -> CertPair:
    return CertPair(
        cert_pem=cert.public_bytes(serialization.Encoding.PEM).decode(),
        key_pem=key.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.PKCS8,
            encryption_algorithm=serialization.NoEncryption(),
        ).decode(),
    )


def issue_ca(common_name: str = "spatiamed-test-ca") -> CertPair:
    key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
    name = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, common_name)])
    now = datetime.datetime.now(datetime.timezone.utc)
    cert = (
        x509.CertificateBuilder()
        .subject_name(name)
        .issuer_name(name)
        .public_key(key.public_key())
        .serial_number(x509.random_serial_number())
        .not_valid_before(now - datetime.timedelta(minutes=5))
        .not_valid_after(now + datetime.timedelta(days=1))
        .add_extension(x509.BasicConstraints(ca=True, path_length=None), critical=True)
        .sign(key, hashes.SHA256())
    )
    return _serialize(cert, key)


def issue_leaf(ca: CertPair, common_name: str, *, server: bool = False) -> CertPair:
    ca_key = serialization.load_pem_private_key(ca.key_pem.encode(), password=None)
    ca_cert = x509.load_pem_x509_certificate(ca.cert_pem.encode())
    key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
    now = datetime.datetime.now(datetime.timezone.utc)
    builder = (
        x509.CertificateBuilder()
        .subject_name(x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, common_name)]))
        .issuer_name(ca_cert.subject)
        .public_key(key.public_key())
        .serial_number(x509.random_serial_number())
        .not_valid_before(now - datetime.timedelta(minutes=5))
        .not_valid_after(now + datetime.timedelta(days=1))
        .add_extension(x509.BasicConstraints(ca=False, path_length=None), critical=True)
    )
    if server:
        # httpx verifies the hostname, so the server certificate needs a SAN
        # covering the address the test connects to.
        builder = builder.add_extension(
            x509.SubjectAlternativeName(
                [x509.DNSName("localhost"), x509.IPAddress(ipaddress.ip_address("127.0.0.1"))]
            ),
            critical=False,
        )
    assert isinstance(ca_key, RSAPrivateKey)
    return _serialize(builder.sign(ca_key, hashes.SHA256()), key)
```

```python
"""End-to-end TLS tests against a real TLS server.

These deliberately avoid respx. A mocked transport cannot tell us whether a
client certificate was presented or whether a private CA was trusted, and
mock-only coverage is exactly what hid the earlier defects in this adapter.
"""

from __future__ import annotations

import asyncio
import contextlib
import os
import ssl
import tempfile
from collections.abc import AsyncIterator

import pytest

from sm_common.integrations.adapters.fhir_r4 import FhirR4Adapter
from sm_common.integrations.tls import TlsConfig
from tests.integrations.certs import CertPair, issue_ca, issue_leaf

pytestmark = pytest.mark.asyncio


class _TlsServer:
    """A minimal HTTPS server that records the client certificate it saw."""

    def __init__(self, ctx: ssl.SSLContext) -> None:
        self._ctx = ctx
        self.peer_common_names: list[str | None] = []
        self._server: asyncio.AbstractServer | None = None

    async def start(self) -> int:
        self._server = await asyncio.start_server(self._handle, "127.0.0.1", 0, ssl=self._ctx)
        return int(self._server.sockets[0].getsockname()[1])

    async def stop(self) -> None:
        if self._server is not None:
            self._server.close()
            await self._server.wait_closed()

    async def _handle(self, reader: asyncio.StreamReader, writer: asyncio.StreamWriter) -> None:
        peer = writer.get_extra_info("peercert")
        cn: str | None = None
        if peer:
            for rdn in peer.get("subject", ()):
                for key, value in rdn:
                    if key == "commonName":
                        cn = value
        self.peer_common_names.append(cn)
        try:
            await reader.readuntil(b"\r\n\r\n")
        except (asyncio.IncompleteReadError, asyncio.LimitOverrunError):
            pass
        body = b'{"resourceType":"CapabilityStatement","status":"active"}'
        writer.write(
            b"HTTP/1.1 200 OK\r\nContent-Type: application/fhir+json\r\n"
            b"Content-Length: " + str(len(body)).encode() + b"\r\n\r\n" + body
        )
        await writer.drain()
        writer.close()


@contextlib.asynccontextmanager
async def _serve(
    ca: CertPair, server: CertPair, *, require_client_cert: bool
) -> AsyncIterator[tuple[str, _TlsServer]]:
    ctx = ssl.create_default_context(ssl.Purpose.CLIENT_AUTH)
    fd, path = tempfile.mkstemp(suffix=".pem")
    try:
        with os.fdopen(fd, "w") as handle:
            handle.write(server.cert_pem)
            handle.write(server.key_pem)
        ctx.load_cert_chain(certfile=path)
    finally:
        os.unlink(path)
    if require_client_cert:
        ctx.verify_mode = ssl.CERT_REQUIRED
        ctx.load_verify_locations(cadata=ca.cert_pem)
    srv = _TlsServer(ctx)
    port = await srv.start()
    try:
        yield f"https://localhost:{port}", srv
    finally:
        await srv.stop()
```

```bash
#!/usr/bin/env bash
# Generate a throwaway CA, server certificate and client certificate for the
# mTLS harness. Output is gitignored: CI runs full-history gitleaks and a
# committed key is a finding forever. Re-run freely; certs last 30 days.
set -euo pipefail
cd "$(dirname "$0")"
mkdir -p certs
cd certs

openssl req -x509 -newkey rsa:2048 -nodes -keyout ca.key -out ca.crt \
  -days 30 -subj "/CN=spatiamed-harness-ca"

openssl req -newkey rsa:2048 -nodes -keyout server.key -out server.csr \
  -subj "/CN=localhost"
printf "subjectAltName=DNS:localhost,IP:127.0.0.1\n" > server.ext
openssl x509 -req -in server.csr -CA ca.crt -CAkey ca.key -CAcreateserial \
  -out server.crt -days 30 -extfile server.ext

openssl req -newkey rsa:2048 -nodes -keyout client.key -out client.csr \
  -subj "/CN=spatiamed-client"
openssl x509 -req -in client.csr -CA ca.crt -CAkey ca.key -CAcreateserial \
  -out client.crt -days 30

rm -f server.csr client.csr server.ext
chmod 600 ./*.key
echo "Wrote harness/openemr/certs/{ca,server,client}.{crt,key}"
```

```python
"""Prove the adapter's mTLS path against a server that really demands a cert.

Run:
    ./gen_mtls_certs.sh
    docker compose up -d
    uv run --no-sync python harness/openemr/verify_mtls.py

Expects BOTH outcomes: refused without a client certificate, accepted with
one. A test that only checks the happy path cannot tell mTLS from plain TLS.
"""

from __future__ import annotations

import asyncio
import pathlib
import sys

from sm_common.integrations.adapters.fhir_r4 import FhirR4Adapter
from sm_common.integrations.tls import TlsConfig

CERTS = pathlib.Path(__file__).parent / "certs"
BASE_URL = "https://localhost:9443/apis/default/fhir"


def _read(name: str) -> str:
    path = CERTS / name
    if not path.exists():
        sys.exit(f"missing {path} — run ./gen_mtls_certs.sh first")
    return path.read_text()


async def main() -> int:
    ca_pem = _read("ca.crt")

    without = FhirR4Adapter(
        base_url=BASE_URL,
        auth_scheme="bearer",
        auth_cfg={"bearer_token": "unused"},
        tls=TlsConfig(ca_bundle_pem=ca_pem),
    )
    refused = await without.health_check()
    await without.close()
    print(f"no client cert  -> healthy={refused.healthy} ({refused.message})")

    with_cert = FhirR4Adapter(
        base_url=BASE_URL,
        auth_scheme="bearer",
        auth_cfg={"bearer_token": "unused"},
        tls=TlsConfig(
            ca_bundle_pem=ca_pem,
            client_cert_pem=_read("client.crt"),
            client_key_pem=_read("client.key"),
        ),
    )
    accepted = await with_cert.health_check()
    await with_cert.close()
    print(f"with client cert-> healthy={accepted.healthy} ({accepted.message})")

    if refused.healthy:
        print("FAIL: the server accepted a connection with no client certificate")
        return 1
    if "connection error" not in (refused.message or ""):
        # An HTTP status (401, 302 to the OpenEMR login) means we reached the
        # application, so TLS did NOT refuse us and this proves nothing.
        print(f"FAIL: expected a TLS-level refusal, got: {refused.message}")
        return 1
    if not accepted.healthy:
        print(f"FAIL: the server rejected our client certificate: {accepted.message}")
        return 1
    print("PASS: mTLS verified in both directions")
    return 0


if __name__ == "__main__":
    raise SystemExit(asyncio.run(main()))
```
